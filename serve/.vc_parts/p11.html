// ---------- populate world ---------- function populate(){ // ambient pedestrians for(let i=0;i<130;i++){ const p = randomStreetPos(); spawnPed(clamp(p.x+rnd(-90,90),40,OCEAN_X-40), clamp(p.y+rnd(-90,90),40,WORLD.h-40)); } // gang members in Little Havana / Haiti for(let i=0;i<18;i++) spawnPed(rnd(350,1900), rnd(700,4300), { gang:true, hp:80, speed:90, col:'#ff006e', dropMoney:80 }); // traffic for(let i=0;i<46;i++){ const p = randomStreetPos(); spawnVehicle(p.x, p.y, { a:p.a, driver:'ai', targetSpeed: rnd(150,250) }); } // parked cars for(let i=0;i<34;i++){ const p = randomStreetPos(); spawnVehicle(p.x + rnd(-24,24), p.y + rnd(-24,24), { a:p.a + rnd(-0.1,0.1), driver:null }); } // a fast car near spawn for the player spawnVehicle(770, 640, { a: 0.4, driver:null, model: CAR_MODELS[0] }); scatterPickups(); setupStage(); if(curStage()) msg("▶ " + curMission().brief, 4000); } // ---------- day/night ---------- function updateDayNight(dt){ dayCol.t += dt/150; // full cycle = 150s if(dayCol.t > 1) dayCol.t -= 1; } function drawNightOverlay(){ const t = dayCol.t; // darkness curve: 0 at noon-ish, peaks at night let dark = 0; if(t < 0.15) dark = 1 - t/0.15; else if(t < 0.55) dark = 0; else if(t < 0.75) dark = (t-0.55)/0.20; else dark = 1; dark *= 0.62; if(dark <= 0.01) return; // bluish night tint ctx.fillStyle = `rgba(8,10,46,${dark})`; ctx.fillRect(0,0,W,H); // street lamp glows for(const p of props){ if(p.type!=='lamp') continue; const [sx,sy] = worldToScreen(p.x,p.y); if(sx<-80||sy<-80||sx>W+80||sy>H+80) continue; const g = ctx.createRadialGradient(sx,sy,4,sx,sy,70); g.addColorStop(0, `rgba(253,230,138,${0.30*dark})`); g.addColorStop(1, 'rgba(253,230,138,0)'); ctx.fillStyle = g; ctx.beginPath(); ctx.arc(sx,sy,70,0,TAU); ctx.fill(); } // headlight cones for(const v of vehicles){ if(v.exploded) continue; const [sx,sy] = worldToScreen(v.x, v.y); if(sx<-160||sy<-160||sx>W+160||sy>H+160) continue; ctx.save(); ctx.translate(sx,sy); ctx.rotate(v.a); const g = ctx.createLinearGradient(20,0,140,0); g.addColorStop(0, `rgba(255,250,200,${0.30*dark})`); g.addColorStop(1,'rgba(255,250,200,0)'); ctx.fillStyle = g; ctx.beginPath(); ctx.moveTo(20,-8); ctx.lineTo(150,-46); ctx.lineTo(150,46); ctx.lineTo(20,8); ctx.closePath(); ctx.fill(); ctx.restore(); } } // ---------- main loop ---------- let started = false, lastT = performance.now(); function frame(now){ const dt = Math.min(0.05, (now - lastT)/1000); lastT = now; if(started){ updateDayNight(dt); updatePlayer(dt); updatePeds(dt); updateVehicles(dt); updateBullets(dt); updatePickups(dt); updateWanted(dt); checkStage(dt); } // ---- draw ---- ctx.clearRect(0,0,W,H); drawGround(); drawDecals(); drawBuildings(); drawProps(); drawPalms(); drawPickups(); drawMissionMarker(); // depth-sorted entities const ents = []; for(const v of vehicles) ents.push({y:v.y, t:'v', o:v}); for(const p of peds) ents.push({y:p.y, t:'p', o:p}); if(!player.inCar) ents.push({y:player.y, t:'pl', o:player}); ents.sort((a,b)=>a.y-b.y); for(const e of ents){ if(e.t==='v') drawVehicle(e.o); else if(e.t==='p') drawPed(e.o); else drawPlayer(); } drawBullets(); drawParticles(); drawNightOverlay(); if(started) drawHUD(); requestAnimationFrame(frame); } requestAnimationFrame(frame); // ---------- start ---------- document.getElementById('startBtn').addEventListener('click', ()=>{ initAudio(); const t = document.getElementById('title'); t.style.opacity = 0; setTimeout(()=> t.style.display='none', 600); started = true; populate(); msg("🌴 Welcome to Vice City, 1986", 3500); setTimeout(()=>{ if(started) msg(curMission().name + " — " + curMission().brief, 5000); }, 3800); });
